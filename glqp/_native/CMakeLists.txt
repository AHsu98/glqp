cmake_minimum_required(VERSION 3.16)
project(glqp_presolve LANGUAGES CXX)
add_library(glqp_presolve SHARED presolve.cpp)
target_compile_features(glqp_presolve PUBLIC cxx_std_17)
target_include_directories(glqp_presolve PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(glqp_presolve PRIVATE -ffp-contract=off -Wall -Wextra -Wpedantic)
endif()
add_executable(glqp-presolve presolve_cli.cpp)
target_link_libraries(glqp-presolve PRIVATE glqp_presolve)
