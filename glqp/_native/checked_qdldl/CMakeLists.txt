cmake_minimum_required(VERSION 3.16)
project(glqp_checked_qdldl LANGUAGES C CXX)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# The original package uses float64 and long-long indices on 64-bit Linux.
set(QDLDL_FLOAT_TYPE "double")
set(QDLDL_INT_TYPE "long long")
set(QDLDL_INT_TYPE_MAX "LLONG_MAX")
set(QDLDL_BOOL_TYPE "unsigned char")
set(QDLDL_LONG 1)
set(QDLDL_VERSION_MAJOR 0)
set(QDLDL_VERSION_MINOR 1)
set(QDLDL_VERSION_PATCH 9)
configure_file(c/qdldl/configure/qdldl_types.h.in generated/qdldl_types.h)
configure_file(c/qdldl/configure/qdldl_version.h.in generated/qdldl_version.h)
file(GLOB AMD_SOURCES CONFIGURE_DEPENDS c/amd/src/*.c)
add_library(_glqp_qdldl_checked MODULE cpp/qdldl.cpp cpp/wrapper.cpp c/qdldl/src/qdldl.c ${AMD_SOURCES})
target_include_directories(_glqp_qdldl_checked PRIVATE c c/amd/include c/qdldl/include
    "${CMAKE_CURRENT_BINARY_DIR}/generated" "${PYTHON_INCLUDE_DIR}" "${PYBIND11_INCLUDE_DIR}")
# No fast math, native CPU tuning, or fused contraction: preserve scalar kernels.
target_compile_options(_glqp_qdldl_checked PRIVATE -O3 -fno-fast-math -ffp-contract=off)
set_target_properties(_glqp_qdldl_checked PROPERTIES PREFIX "" SUFFIX "${PYTHON_EXTENSION_SUFFIX}")
